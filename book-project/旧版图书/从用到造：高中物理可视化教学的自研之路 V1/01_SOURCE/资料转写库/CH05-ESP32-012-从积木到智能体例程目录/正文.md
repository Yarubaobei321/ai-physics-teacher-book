/**
 * 课程例程库（Mixly 式）：按合集 / 篇章列出可一键载入的积木工程或 MicroPython 代码。
 * Blockly workspace 与 .esp32 导出格式相同；代码例程只写入代码视图，不改 WS 协议。
 */

let __exampleId = 0;
function eid(prefix) {
  __exampleId += 1;
  return `${prefix || 'ex'}_${__exampleId}`;
}
function numberBlock(n) {
  return { type: 'math_number', id: eid('num'), fields: { NUM: n } };
}
function textBlock(text) {
  return { type: 'text', id: eid('text'), fields: { TEXT: text } };
}
function block(type, opts) {
  return Object.assign({ type, id: eid(type) }, opts || {});
}
function input(blockValue) {
  return { block: blockValue };
}
function shadow(blockValue) {
  return { shadow: blockValue };
}
function chain(blocks) {
  const items = blocks.filter(Boolean).map((b) => JSON.parse(JSON.stringify(b)));
  for (let i = items.length - 2; i >= 0; i--) items[i].next = { block: items[i + 1] };
  return items[0] || null;
}
function workspace(root) {
  const blocks = Array.isArray(root) ? [chain(root)] : [root];
  return { blocks: { languageVersion: 0, blocks: blocks.filter(Boolean) } };
}
function forever(blocks) {
  const body = chain(blocks);
  return block('esp32_forever', body ? { inputs: { DO: input(body) } } : {});
}
function print(valueBlock) {
  return block('esp32_print', { inputs: { VALUE: input(valueBlock) } });
}
function printLabel(label, valueBlock) {
  return block('esp32_print_labeled', {
    inputs: { LABEL: input(textBlock(label)), VALUE: input(valueBlock) },
  });
}
function delayMs(ms) {
  return block('esp32_delay', { inputs: { MS: input(numberBlock(ms)) } });
}
function delayS(s) {
  return block('esp32_delay_s', { inputs: { S: input(numberBlock(s)) } });
}
function ifBlock(condition, thenBlocks, elseBlocks) {
  const opts = {
    inputs: {
      IF0: input(condition),
      DO0: input(chain(thenBlocks)),
    },
  };
  if (elseBlocks && elseBlocks.length) {
    opts.extraState = { hasElse: true };
    opts.inputs.ELSE = input(chain(elseBlocks));
  }
  return block('controls_if', opts);
}
function compare(op, left, right) {
  return block('logic_compare', {
    fields: { OP: op },
    inputs: { A: input(left), B: input(right) },
  });
}
function logicAnd(left, right) {
  return block('logic_operation', {
    fields: { OP: 'AND' },
    inputs: { A: input(left), B: input(right) },
  });
}
function digitalWrite(pin, state) {
  return block('esp32_digital_write', { fields: { PIN: String(pin), STATE: state ? '1' : '0' } });
}
function led(state) {
  return block('esp32_led_builtin', { fields: { STATE: state ? '1' : '0' } });
}
function pwm(pin, duty) {
  const value = typeof duty === 'number' ? numberBlock(duty) : duty;
  return block('esp32_pwm_write', { fields: { PIN: String(pin) }, inputs: { DUTY: input(value) } });
}
function rgbFill(r, g, b) {
  return block('esp32_rgb_fill', { fields: { PIN: '13', COUNT: 8, R: r, G: g, B: b } });
}
function rgbSet(index, r, g, b) {
  return block('esp32_rgb_set', { fields: { PIN: '13', COUNT: 8, INDEX: index, R: r, G: g, B: b } });
}
function rgbOff() {
  return block('esp32_rgb_off', { fields: { PIN: '13', COUNT: 8 } });
}
function oledLine(textOrBlock, line) {
  const value = typeof textOrBlock === 'string' ? textBlock(textOrBlock) : textOrBlock;
  return block('esp32_oled_text', { fields: { LINE: line }, inputs: { TEXT: input(value) } });
}
function httpGet(url) {
  const value = typeof url === 'string' ? textBlock(url) : url;
  return block('esp32_http_get', { inputs: { URL: input(value) } });
}
function jsonGet(jsonText, key) {
  const value = typeof jsonText === 'string' ? textBlock(jsonText) : jsonText;
  return block('esp32_json_get', { inputs: { JSON: input(value), KEY: input(textBlock(key)) } });
}
function mqttTopic(prefix, device, name) {
  return block('esp32_mqtt_topic', {
    inputs: {
      PREFIX: input(textBlock(prefix)),
      DEVICE: input(textBlock(device)),
      NAME: input(textBlock(name)),
    },
  });
}
function mqttPublish(broker, client, topicBlock, messageBlock) {
  const topic = typeof topicBlock === 'string' ? textBlock(topicBlock) : topicBlock;
  const msg = typeof messageBlock === 'string' ? textBlock(messageBlock) : messageBlock;
  return block('esp32_mqtt_publish', {
    inputs: {
      BROKER: input(textBlock(broker)),
      CLIENT: input(textBlock(client)),
      TOPIC: input(topic),
      MESSAGE: input(msg),
    },
  });
}
function common(meta) {
  return Object.assign({
    collectionId: 'from-blocks-to-agent',
    collectionTitle: '从积木到智能体',
    kind: 'blocks',
  }, meta);
}

  init(listEl, onPick) {
    this.listEl = listEl;
    this.onPick = onPick;
    this.render();
  },

  render() {
    const el = this.listEl;
    if (!el) return;
    el.innerHTML = '';

    const byCollection = new Map();
    const items = this.CATALOG.slice().sort((a, b) => (a.order || 0) - (b.order || 0));
    for (const ex of items) {
      const cid = ex.collectionId || 'default';
      if (!byCollection.has(cid)) {
        byCollection.set(cid, { title: ex.collectionTitle || '课程例程', items: [] });
      }
      byCollection.get(cid).items.push(ex);
    }

    for (const collection of byCollection.values()) {
      const sec = document.createElement('section');
      sec.className = 'examples-section examples-collection';
      const h = document.createElement('h4');
      h.className = 'examples-cat examples-collection-title';
      h.textContent = collection.title;
      sec.appendChild(h);

      let currentPart = '';
      let currentChapter = '';
      for (const ex of collection.items) {
        const part = ex.part || ex.category || '';
        const chapter = ex.chapter || '';
        if (part && part !== currentPart) {
          currentPart = part;
          currentChapter = '';
          const p = document.createElement('div');
          p.className = 'examples-part';
          p.textContent = part;
          sec.appendChild(p);
        }
        if (chapter && chapter !== currentChapter) {
          currentChapter = chapter;
          const c = document.createElement('div');
          c.className = 'examples-chapter';
          c.textContent = chapter;
          sec.appendChild(c);
        }

        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = 'example-card focus-ring';
        const title = document.createElement('span');
        title.className = 'example-title';
        title.textContent = ex.title;
        const desc = document.createElement('span');
        desc.className = 'example-desc';
        desc.textContent = ex.desc || '';
        const meta = document.createElement('span');
        meta.className = 'example-meta';
        meta.textContent = ex.kind === 'code' ? 'MicroPython 代码' : '积木工程';
        btn.append(title, desc, meta);
        btn.onclick = () => this.onPick && this.onPick(ex);
        sec.appendChild(btn);
      }
      el.appendChild(sec);
    }
  },

  get(id) {
    return this.CATALOG.find((e) => e.id === id);
  },
};
