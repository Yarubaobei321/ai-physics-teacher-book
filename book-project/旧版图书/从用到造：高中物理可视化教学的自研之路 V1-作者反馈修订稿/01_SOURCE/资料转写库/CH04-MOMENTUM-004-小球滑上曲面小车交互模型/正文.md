    function clamp(value, min, max) {
      return Math.max(min, Math.min(max, value));
    }

    function getXScale() {
      return (geom.rightFootX - geom.leftFootX) / geom.curveWidthM;

	    function readParams() {
	      const v = Number(els.speedRange.value);
	      const g = state.g;
	      const height = Number(els.heightRange.value);
	      const ballMass = Number(els.ballMassRange.value);
	      const cartMass = Number(els.cartMassRange.value);
	      const totalMass = ballMass + cartMass;
	      const cartShare = cartMass / totalMass;
	      const ballShare = ballMass / totalMass;
	      const critical = Math.sqrt((2 * g * height * totalMass) / cartMass);
	      const reachesRight = v + 1e-9 >= critical;
	      const initialEnergy = 0.5 * ballMass * v * v;
	      return { v, g, height, ballMass, cartMass, totalMass, cartShare, ballShare, critical, reachesRight, initialEnergy };
	    }

	    function velocityAtSurface(surface, params, direction = 1) {
	      const numerator = Math.max(0, params.cartShare * params.v * params.v - 2 * params.g * surface.heightM);
	      const relativeAbs = Math.sqrt(numerator / (params.cartShare + surface.slope * surface.slope));
	      const relative = direction * relativeAbs;
	      const cart = params.ballShare * (params.v - relative);
	      const ballHorizontal = cart + relative;
	      const ballVertical = surface.slope * relative;
	      const ballTotal = Math.hypot(ballHorizontal, ballVertical);
	      return { ball: ballHorizontal, cart, rel: relative, relAbs: relativeAbs, ballVertical, ballTotal };
	    }

	    function energyAt(surface, velocity, params) {
	      const cartKinetic = 0.5 * params.cartMass * velocity.cart * velocity.cart;
	      const ballKinetic = 0.5 * params.ballMass * velocity.ballTotal * velocity.ballTotal;
	      const potential = params.ballMass * params.g * surface.heightM;
	      return { cartKinetic, ballKinetic, potential, total: params.initialEnergy };
	    }

	    function findTurningQ(params) {
	      const targetHeight = params.cartShare * params.v * params.v / (2 * params.g);
	      let lo = 0;
	      let hi = geom.curveWidthM / 2;
	      for (let i = 0; i < 48; i += 1) {
	        const mid = (lo + hi) / 2;
	        const surface = trackSurfaceAtQ(mid, params);
	        if (surface.heightM < targetHeight) lo = mid;
	        else hi = mid;
	      }
	      return (lo + hi) / 2;
	    }

    function trackSurfaceAtQ(q, params) {
      const xScale = getXScale();
      const qClamped = clamp(q, 0, geom.curveWidthM);
      const s = qClamped / geom.curveWidthM;
      const onCurve = qClamped >= 0 && qClamped <= geom.curveWidthM;
      const centerDistance = Math.abs(1 - 2 * s);
      const curveCore = onCurve ? 1 - Math.pow(centerDistance, 1.5) : 0;
      const bump = curveCore * curveCore;
      let slope = 0;
      if (onCurve && centerDistance > 1e-9) {
        const direction = s < 0.5 ? 1 : -1;
        slope = direction * 6 * params.height * curveCore * Math.sqrt(centerDistance) / geom.curveWidthM;
      }
      const visualSlope = -slope * geom.pxPerM / xScale;
      const normalLen = Math.hypot(visualSlope, 1);
      return {
        q: qClamped,
        x: geom.leftFootX + qClamped * xScale,
        y: geom.deckY - params.height * geom.pxPerM * bump,
        slope,
        normalX: visualSlope / normalLen,
        normalY: -1 / normalLen,
        heightM: params.height * bump
      };
    }

    function makeTrackPath(params) {
      const y = geom.deckY;
      const points = [];
      for (let i = 0; i <= 64; i += 1) {
        const q = geom.curveWidthM * (i / 64);
        const surface = trackSurfaceAtQ(q, params);
        points.push(`${surface.x.toFixed(1)} ${surface.y.toFixed(1)}`);
      }
      return `M ${geom.leftEdge} ${y} L ${geom.leftFootX} ${y} L ${points.join(" L ")} L ${geom.rightEdge} ${y}`;
    }

	    function buildMotionProfile(params) {
	      const samples = [];
	      let time = 0;
	      let cartShiftM = 0;
	      let previous = null;
		      const count = 900;
	      const qEnd = params.reachesRight ? geom.curveWidthM : findTurningQ(params);
	      const totalDistance = params.reachesRight ? qEnd : 2 * qEnd;

	      function pushSample(q, direction, pathDistance) {
	        const surface = trackSurfaceAtQ(q, params);
	        const velocity = velocityAtSurface(surface, params, direction);
	        if (previous) {
	          const dq = Math.abs(q - previous.q);
	          const averageRelative = Math.max((previous.velocity.relAbs + velocity.relAbs) / 2, 1e-6);
	          const dt = dq / averageRelative;
	          time += dt;
	          cartShiftM += ((previous.velocity.cart + velocity.cart) / 2) * dt;
	        }
	        const sample = {
	          q,
	          time,
	          surface,
	          velocity,
	          cartShiftM,
	          pathProgress: totalDistance > 0 ? pathDistance / totalDistance : 0,
	          direction
	        };
	        samples.push(sample);
	        previous = sample;
	      }

	      for (let i = 0; i <= count; i += 1) {
	        const q = qEnd * (i / count);
	        pushSample(q, 1, q);
	      }

	      if (!params.reachesRight) {
	        for (let i = 1; i <= count; i += 1) {
	          const q = qEnd * (1 - i / count);
	          pushSample(q, -1, qEnd + qEnd * (i / count));
	        }
	      }

	      const maxSpeed = samples.reduce((max, sample) => Math.max(max, sample.velocity.ballTotal, Math.abs(sample.velocity.cart)), params.v);
	      return { samples, totalTime: time, qEnd, reachesRight: params.reachesRight, maxSpeed };
	    }

	    function ensureProfile(params) {
	      const key = `${params.v.toFixed(4)}-${params.height.toFixed(4)}-${params.ballMass.toFixed(4)}-${params.cartMass.toFixed(4)}`;
	      if (!state.profile || state.profile.key !== key) {
	        state.profile = buildMotionProfile(params);
	        state.profile.key = key;
      }
      return state.profile;
    }

    function interpolateBy(items, value, field) {
      if (value <= items[0][field]) return items[0];
      const last = items[items.length - 1];
      if (value >= last[field]) return last;
      let lo = 0;
      let hi = items.length - 1;
      while (hi - lo > 1) {
        const mid = Math.floor((lo + hi) / 2);
        if (items[mid][field] <= value) lo = mid;
        else hi = mid;
      }
      const a = items[lo];
      const b = items[hi];
      const ratio = (value - a[field]) / (b[field] - a[field] || 1);
      return { a, b, ratio };
    }

	    function sampleAtTime(profile, params, time) {
	      const result = interpolateBy(profile.samples, time, "time");
	      if (!result.a) return result;
	      const q = result.a.q + (result.b.q - result.a.q) * result.ratio;
	      const cartShiftM = result.a.cartShiftM + (result.b.cartShiftM - result.a.cartShiftM) * result.ratio;
	      const pathProgress = result.a.pathProgress + (result.b.pathProgress - result.a.pathProgress) * result.ratio;
	      const rel = result.a.velocity.rel + (result.b.velocity.rel - result.a.velocity.rel) * result.ratio;
	      const direction = rel < 0 || (Math.abs(rel) < 1e-8 && result.b.direction < 0) ? -1 : 1;
	      const surface = trackSurfaceAtQ(q, params);
	      const velocity = velocityAtSurface(surface, params, direction);
	      return { q, time, surface, velocity, cartShiftM, pathProgress, direction };
	    }

	    function timeAtProgress(profile, progress) {
	      const result = interpolateBy(profile.samples, progress, "pathProgress");
	      if (!result.a) return result.time;
	      return result.a.time + (result.b.time - result.a.time) * result.ratio;
	    }

		      const potentialW = clamp(energy.potential * energyScale, 0, 118);
		      els.cartKineticBar.setAttribute("width", cartEnergyW.toFixed(1));
		      els.ballKineticBar.setAttribute("width", ballEnergyW.toFixed(1));
		      els.potentialBar.setAttribute("width", potentialW.toFixed(1));
	      els.energyLayer.style.display = els.showEnergy.checked ? "" : "none";
	      els.externalEnergyPanel.style.display = els.showEnergy.checked ? "" : "none";
	      const panelEnergyItems = [
	        [els.panelCartKineticBar, els.panelCartKineticText, energy.cartKinetic],
	        [els.panelBallKineticBar, els.panelBallKineticText, energy.ballKinetic],
	        [els.panelPotentialBar, els.panelPotentialText, energy.potential]
	      ];
	      for (const [bar, label, value] of panelEnergyItems) {
	        const percent = clamp(value / Math.max(params.initialEnergy, 0.001) * 100, 0, 100);
	        bar.style.width = `${percent.toFixed(1)}%`;
	        label.textContent = `${percent.toFixed(1)}%`;
	      }

	      els.ballSpeedRead.textContent = `${vNow.ballTotal.toFixed(2)} m/s`;
	      els.cartSpeedRead.textContent = `${Math.abs(vNow.cart).toFixed(2)} m/s`;
	      els.momentumRead.textContent = `${(params.cartMass * vNow.cart).toFixed(2)} kg·m/s`;
	      els.timeReadout.textContent = `t = ${state.modelTime.toFixed(2)} s`;
	      drawVelocityGraph(profile, params, state.modelTime);

	      if (state.progress < 0.08) {
	        els.stateRead.textContent = "滑上左端";
	        els.caption.textContent = "小球贴着曲面左端进入，系统总水平动量来自小球。";

	        els.vtPhaseText.textContent = "初始：小球最快，小车静止";
	      } else if (Math.abs(state.progress - 0.5) < 0.05) {
	        if (params.reachesRight) {
	          els.stateRead.textContent = "到达最高点";
	          els.caption.textContent = params.v > params.critical + 0.02 ? "越过最高点：小球仍有相对速度，将从右侧下滑。" : "临界到顶：小球相对小车速度瞬时为零。";
	          setFrameCaption(params.v > params.critical + 0.02 ? "越过最高点" : "临界到顶", params.v > params.critical + 0.02 ? "小球将从右侧下滑" : "相对速度瞬时为零");
	          els.vtPhaseText.textContent = params.v > params.critical + 0.02 ? "最高点：仍能继续向右" : "最高点：临界共速";
	        } else {
	          els.stateRead.textContent = "到达转折点";
	          els.caption.textContent = "低于临界：小球相对小车速度变为零，随后从左侧滑回。";
	          setFrameCaption("到达转折点", "随后从左侧滑回");
	          els.vtPhaseText.textContent = "转折点：相对速度为零";
	        }
	      } else if (state.progress > 0.92) {
	        els.stateRead.textContent = params.reachesRight ? "右端滑下" : "左端滑下";
	        els.caption.textContent = params.reachesRight
	          ? "从右端滑下：小球速度重新变大，小车速度由质量比决定。"
	          : "从左端滑下：小球把一部分水平动量传给小车。";
	        setFrameCaption(params.reachesRight ? "从右端滑下" : "从左端滑下", params.reachesRight ? "速度分配由质量比决定" : "动量转移给小车");
	        els.vtPhaseText.textContent = params.reachesRight ? "右端：小球重新变快" : "左端：小车获得速度";
	      } else if (state.progress < 0.5) {
	        els.stateRead.textContent = "上滑";
	        els.caption.textContent = "上滑时小球减速，小车加速；动能一部分转化为重力势能。";
	        setFrameCaption("上滑：小球减速", "小车加速，重力势能增加");
	        els.vtPhaseText.textContent = "上滑：小球减速，小车加速";

	        els.stateRead.textContent = params.reachesRight ? "右侧下滑" : "返回左侧";

	        els.vtPhaseText.textContent = params.reachesRight ? "下滑：小球加速" : "返回：相对速度向左";

    function setPlaying(next) {
      state.playing = next;
      state.lastTs = null;
      els.playBtn.textContent = next ? "暂停" : "播放";
    }

	    function reset() {
	      setPlaying(false);
	      state.progress = 0;
	      state.q = 0;
	      state.modelTime = 0;
	      updateScene();
	    }

	    function resetModelOnly() {
	      state.progress = 0;
	      state.q = 0;
	      state.modelTime = 0;
	      state.profile = null;
	      setPlaying(false);
	      updateScene();
	    }

	    function chooseCase(type) {
	      const params = readParams();
	      const min = Number(els.speedRange.min);
	      const max = Number(els.speedRange.max);
	      const target = type === "slow" ? params.critical * 0.72 : params.critical * 1.12;
	      els.speedRange.value = clamp(target, min, max).toFixed(1);
	      resetModelOnly();
	    }

    function animate(ts) {
      if (!state.playing) return;
      if (state.lastTs == null) state.lastTs = ts;
      const dt = ts - state.lastTs;
      state.lastTs = ts;
      const params = readParams();
      const profile = ensureProfile(params);
      state.modelTime += (dt / 1000) * Number(els.playSpeedRange.value);
	      if (state.modelTime >= profile.totalTime) {
	        state.modelTime = profile.totalTime;
	        state.progress = 1;
	        setPlaying(false);
	      }
	      updateScene();
	      requestAnimationFrame(animate);
    }

    els.playBtn.addEventListener("click", () => {
      setPlaying(!state.playing);
      if (state.playing) requestAnimationFrame(animate);
    });

    els.resetBtn.addEventListener("click", reset);

      setPlaying(false);
	      const params = readParams();
	      const profile = ensureProfile(params);
		      state.modelTime = clamp(state.modelTime + STEP_SECONDS, 0, profile.totalTime);
	      updateScene();
	    });

	        setPlaying(false);
	        const params = readParams();
	        const profile = ensureProfile(params);
	        state.progress = Number(button.dataset.jump);

	        updateScene();
	      });
	    });
