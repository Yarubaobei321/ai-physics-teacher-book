'use strict';

/**
 * ESP32Blockly 中央服务器（教师 Mac 上运行）。
 *
 * 职责：
 *   1. 托管学生端 PWA（/）与教师管理后台（/admin）。
 *   2. 一条 WebSocket 通道 (/ws) 同时服务 ESP32 设备与浏览器客户端，
 *      靠首条消息的 type 区分身份。
 *   3. 按设备号路由：把学生上传的代码准确转发给对应 ESP32，并回传串口输出。
 *   4. 校验设备编号是否在名册内（防止操作不存在的编号）。绑定只用编号，不做密钥认证。
 *
 * 隐私：只记录"在线状态"和"上传事件元数据"(时间/设备/字节数/哈希)，
 *       绝不落盘代码正文，绝不记录积木结构。
 *
 * 部署模式（部署无关，二选一）：
 *   A. Cloudflare Tunnel 后端：直接 HTTP，Cloudflare 负责 TLS（推荐）。
 *      PORT=8080 node server.js
 *   B. 本地 HTTPS（mkcert）：提供证书后自动启用 HTTPS。
 *      TLS_KEY=./certs/key.pem TLS_CERT=./certs/cert.pem PORT=8443 node server.js
 */

const fs = require('fs');
const http = require('http');
const https = require('https');
const os = require('os');
const path = require('path');
const crypto = require('crypto');
const express = require('express');
const { WebSocketServer } = require('ws');
const { Registry } = require('./lib/registry');

const PORT = Number(process.env.PORT || 8080);
const HOST = process.env.HOST || '0.0.0.0';
const TLS_KEY = process.env.TLS_KEY;
const TLS_CERT = process.env.TLS_CERT;
const MAX_CODE_BYTES = Number(process.env.MAX_CODE_BYTES || 64 * 1024); // 64KB 上限
const UPLOAD_LOG = path.join(__dirname, 'data', 'uploads.log'); // 仅元数据

const registry = new Registry();

// ---------------------------------------------------------------------------

app.use(express.json({ limit: '1mb' }));

  next();
});

app.use('/admin', express.static(path.join(__dirname, 'admin')));

app.get('/api/health', (_req, res) => res.json({ ok: true, ts: Date.now() }));

function localIpv4Addresses() {
  const result = [];
  const nets = os.networkInterfaces();
  for (const [name, entries] of Object.entries(nets)) {
    for (const entry of entries || []) {
      if (entry.family !== 'IPv4' || entry.internal) continue;
      result.push({ name, address: entry.address });
    }
  }
  return result.sort((a, b) => {
    const aw = /^en|^wl|^wi/i.test(a.name) ? 0 : 1;
    const bw = /^en|^wl|^wi/i.test(b.name) ? 0 : 1;
    return aw - bw || a.name.localeCompare(b.name) || a.address.localeCompare(b.address);
  });
}

app.get('/api/network-info', (req, res) => {
  const secure = req.secure || req.headers['x-forwarded-proto'] === 'https';
  const httpProto = secure ? 'https' : 'http';
  const wsProto = secure ? 'wss' : 'ws';
  const hostHeader = req.headers.host || `localhost:${PORT}`;
  const portPart = hostHeader.includes(':') ? ':' + hostHeader.split(':').pop() : '';
  const addresses = localIpv4Addresses().map((item) => ({
    ...item,
    appUrl: `${httpProto}://${item.address}${portPart || ':' + PORT}/`,
    wsUrl: `${wsProto}://${item.address}${portPart || ':' + PORT}/ws`,
  }));
  res.json({
    ok: true,
    currentHost: hostHeader,
    currentAppUrl: `${httpProto}://${hostHeader}/`,
    currentWsUrl: `${wsProto}://${hostHeader}/ws`,
    addresses,
  });
});

// 有线烧录页用：MAC → 编号 映射（判断新板/旧板）。无密钥，只含 MAC 与编号。

// 有线烧录页用：白名单提供固件应用文件，供浏览器经 raw-REPL 写入设备。
// 仅暴露这几个文件，绝不暴露 firmware/configs/（含 WiFi 密码）。
const FW_DIR = path.join(__dirname, '..', 'firmware');
const FW_ASSET_WHITELIST = new Set([
  'boot.py', 'main.py', 'supervisor.py', 'ble_transport.py', 'wifi.py', 'ws_client.py',
]);
app.get('/firmware-assets', (_req, res) => res.json([...FW_ASSET_WHITELIST]));
app.get('/firmware-assets/:name', (req, res) => {
  const name = req.params.name;
  if (!FW_ASSET_WHITELIST.has(name)) return res.status(404).send('not allowed');
  const file = path.join(FW_DIR, name);
  if (!fs.existsSync(file)) return res.status(404).send('not found');
  res.type('text/plain; charset=utf-8').send(fs.readFileSync(file, 'utf8'));
});

/** 与 gen-roster / provision.sh 同源：教室 WiFi + 服务器地址（不含 per-board id） */
function loadProvisionDefaults(req) {
  const host = req.headers.host || 'localhost:8080';
  const secure = req.secure || req.headers['x-forwarded-proto'] === 'https';
  const autoServer = `${secure ? 'wss' : 'ws'}://${host}/ws`;

  if (process.env.WIFI_SSID) {
    return {
      wifi: { ssid: process.env.WIFI_SSID, password: process.env.WIFI_PASS || '' },
      server: process.env.SERVER_WS || autoServer,
      source: 'env',
    };
  }
  const cfgDir = path.join(FW_DIR, 'configs');
  if (fs.existsSync(cfgDir)) {
    const files = fs.readdirSync(cfgDir).filter((f) => f.endsWith('.json')).sort();
    if (files.length) {
      const cfg = JSON.parse(fs.readFileSync(path.join(cfgDir, files[0]), 'utf8'));
      return {
        wifi: cfg.wifi,
        server: cfg.server || autoServer,
        source: 'configs',
      };
    }
  }
  const localCfg = path.join(FW_DIR, 'config.json');
  if (fs.existsSync(localCfg)) {
    const cfg = JSON.parse(fs.readFileSync(localCfg, 'utf8'));
    return {
      wifi: cfg.wifi,
      server: cfg.server || autoServer,
      source: 'config.json',
    };
  }
  return { wifi: null, server: autoServer, source: 'none' };
}

// 有线初始化：读取与 provision.sh 相同的 config（优先 firmware/configs/<id>.json）
app.get('/api/provision-config/:id', (req, res) => {
  const id = String(req.params.id || '').trim().toUpperCase();
  if (!id) return res.status(400).json({ ok: false, error: 'missing_id' });

  const perBoard = path.join(FW_DIR, 'configs', `${id}.json`);
  if (fs.existsSync(perBoard)) {
    try {
      const cfg = JSON.parse(fs.readFileSync(perBoard, 'utf8'));
      return res.json({ ok: true, config: cfg, source: 'configs/' + id });
    } catch (_) {
      return res.status(500).json({ ok: false, error: 'config_parse_fail' });
    }
  }

  const defaults = loadProvisionDefaults(req);
  if (!defaults.wifi || !defaults.wifi.ssid) {
    return res.json({
      ok: false,
      error: 'no_wifi_config',
      hint: '请先运行 gen-roster.js 生成 firmware/configs/，或设置环境变量 WIFI_SSID/WIFI_PASS',
      server: defaults.server,
    });
  }
  res.json({
    ok: true,
    config: { id, label: id, wifi: defaults.wifi, server: defaults.server },
    source: defaults.source,
  });
});

// ---------------------------------------------------------------------------
// HTTP(S) server
// ---------------------------------------------------------------------------
let server;
if (TLS_KEY && TLS_CERT) {
  server = https.createServer(
    { key: fs.readFileSync(TLS_KEY), cert: fs.readFileSync(TLS_CERT) },
    app
  );
  console.log('[server] 已启用 HTTPS（本地证书模式）');
} else {
  server = http.createServer(app);
  console.log('[server] HTTP 模式（适用于 Cloudflare Tunnel 后端）');
}

// ---------------------------------------------------------------------------
// WebSocket：一条 /ws 通道服务两类客户端
// ---------------------------------------------------------------------------
const wss = new WebSocketServer({ server, path: '/ws' });

// 浏览器订阅者：把每个学生 ws 绑定到它已验证的某台设备，用于接收串口回显与状态。
/** @type {Map<string, Set<any>>} deviceId -> Set<studentWs> */
const subscribers = new Map();
/** @type {Set<any>} 管理台 ws 集合 */
const admins = new Set();

function subscribe(deviceId, ws) {
  if (!subscribers.has(deviceId)) subscribers.set(deviceId, new Set());
  subscribers.get(deviceId).add(ws);
}
function unsubscribeAll(ws) {
  for (const set of subscribers.values()) set.delete(ws);
}
function sendToSubscribers(deviceId, msg) {
  const set = subscribers.get(deviceId);
  if (!set) return;
  const data = JSON.stringify(msg);
  for (const ws of set) safeSend(ws, data);
}
function broadcastStatus(deviceId, online, fw) {
  const msg = { type: 'device_status', deviceId, online, fw: fw || '' };
  sendToSubscribers(deviceId, msg);
  const adminData = JSON.stringify(msg);
  for (const a of admins) safeSend(a, adminData);
}
function safeSend(ws, data) {
  try {
    if (ws.readyState === ws.OPEN) ws.send(typeof data === 'string' ? data : JSON.stringify(data));
  } catch (_) {
    /* ignore */
  }
}

function logUpload(deviceId, code) {
  // 隐私：只写元数据，不写代码正文。
  const meta = {
    ts: new Date().toISOString(),
    deviceId,
    bytes: Buffer.byteLength(code, 'utf8'),
    sha256: crypto.createHash('sha256').update(code).digest('hex').slice(0, 12),
  };
  fs.appendFile(UPLOAD_LOG, JSON.stringify(meta) + '\n', () => {});
  console.log(`[upload] ${deviceId} ${meta.bytes}B #${meta.sha256}`);
}

wss.on('connection', (ws, req) => {
  ws.isAlive = true;
  ws.role = null; // 'device' | 'student' | 'admin'
  ws.deviceId = null;
  ws.on('pong', () => { ws.isAlive = true; });

  ws.on('message', (raw) => {
    let msg;
    try {
      msg = JSON.parse(raw.toString());
    } catch {
      return safeSend(ws, { type: 'error', reason: 'bad_json' });
    }
    handleMessage(ws, msg);
  });

  ws.on('close', () => {
    if (ws.role === 'device' && ws.deviceId) {
      registry.setOffline(ws.deviceId);
      broadcastStatus(ws.deviceId, false);
      console.log(`[device] ${ws.deviceId} 离线`);
    }
    if (ws.role === 'admin') admins.delete(ws);
    unsubscribeAll(ws);
  });

  ws.on('error', () => {});
});

function handleMessage(ws, msg) {
  switch (msg.type) {
    // ---- ESP32 设备 ----
    case 'device_register': {
      if (!registry.knows(msg.id)) {
        safeSend(ws, { type: 'register_fail', reason: 'unknown_device' });
        return;
      }
      ws.role = 'device';
      ws.deviceId = msg.id;
      registry.setOnline(msg.id, ws, msg.fw);
      safeSend(ws, { type: 'register_ok', id: msg.id });
      broadcastStatus(msg.id, true, msg.fw);
      console.log(`[device] ${msg.id} 上线 (fw=${msg.fw || '?'})`);
      return;
    }
    case 'output': {
      // ESP32 → 服务器 → 订阅该设备的学生（串口回显）
      if (ws.role !== 'device' || !ws.deviceId) return;
      sendToSubscribers(ws.deviceId, { type: 'output', deviceId: ws.deviceId, data: String(msg.data || '') });
      return;
    }
    case 'run_result': {
      if (ws.role !== 'device' || !ws.deviceId) return;
      sendToSubscribers(ws.deviceId, {
        type: 'run_result', deviceId: ws.deviceId, ok: !!msg.ok, error: msg.error || '',
      });
      return;
    }

    // ---- 学生 PWA ----
    case 'verify': {
      // 只校验“编号存在于名册”。绑定即“我要操作这一台”，不做密钥认证。
      if (!registry.knows(msg.deviceId)) {
        safeSend(ws, { type: 'verify_fail', deviceId: msg.deviceId });
        return;
      }
      ws.role = ws.role === 'admin' ? 'admin' : 'student';
      subscribe(msg.deviceId, ws);
      safeSend(ws, {
        type: 'verify_ok',
        deviceId: msg.deviceId,
        online: registry.isOnline(msg.deviceId),
      });
      return;
    }
    case 'upload': {
      // 按编号路由到对应 ESP32。
      if (!registry.knows(msg.deviceId)) {
        safeSend(ws, { type: 'upload_fail', reason: 'unknown_device' });
        return;
      }
      const code = typeof msg.code === 'string' ? msg.code : '';
      if (!code) return safeSend(ws, { type: 'upload_fail', reason: 'empty_code' });
      if (Buffer.byteLength(code, 'utf8') > MAX_CODE_BYTES) {
        return safeSend(ws, { type: 'upload_fail', reason: 'code_too_large' });
      }
      const deviceWs = registry.getDeviceWs(msg.deviceId);
      if (!deviceWs) return safeSend(ws, { type: 'upload_fail', reason: 'device_offline' });

      logUpload(msg.deviceId, code);
      safeSend(deviceWs, { type: 'run', code, lang: msg.lang || 'micropython' });
      safeSend(ws, { type: 'upload_ack', deviceId: msg.deviceId });
      return;
    }
    case 'stop': {
      if (!registry.knows(msg.deviceId)) {
        return safeSend(ws, { type: 'stop_fail', reason: 'unknown_device' });
      }
      const deviceWs = registry.getDeviceWs(msg.deviceId);
      if (deviceWs) safeSend(deviceWs, { type: 'stop' });
      safeSend(ws, { type: 'stop_ack', deviceId: msg.deviceId });
      return;
    }

    // ---- 管理台 ----
    case 'admin_subscribe': {
      ws.role = 'admin';
      admins.add(ws);
      safeSend(ws, { type: 'admin_snapshot', devices: registry.snapshot() });
      return;
    }

    default:
      safeSend(ws, { type: 'error', reason: 'unknown_type' });
  }
}

// 心跳：30s 检测一次死连接。
const heartbeat = setInterval(() => {
  for (const ws of wss.clients) {
    if (ws.isAlive === false) { ws.terminate(); continue; }
    ws.isAlive = false;
    try { ws.ping(); } catch (_) {}
  }
}, 30000);
wss.on('close', () => clearInterval(heartbeat));

server.listen(PORT, HOST, () => {
  const proto = TLS_KEY && TLS_CERT ? 'https' : 'http';
  console.log(`\n  ESP32Blockly 服务器已启动`);
  console.log(`  学生端 PWA:   ${proto}://<本机IP>:${PORT}/`);
  console.log(`  教师管理台:   ${proto}://<本机IP>:${PORT}/admin`);
  console.log(`  WS 通道:      ${proto === 'https' ? 'wss' : 'ws'}://<本机IP>:${PORT}/ws\n`);
});
