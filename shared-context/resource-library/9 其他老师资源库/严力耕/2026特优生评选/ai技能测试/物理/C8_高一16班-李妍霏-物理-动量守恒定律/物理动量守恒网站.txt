    function bindSliders(){

    function calcR(m){ if(scene==='car') return 1.2+0.0002*m; if(scene==='bullet') return 0.01+0.0002*m; return 0.15+m*0.025; }
    function toPx(x){ const pl=60, pr=canvasW-60; return pl+((x-trackLeft)/(trackRight-trackLeft))*(pr-pl); }
    function trackY(){ return canvasH*0.55; }
    function resizeCanvas(){
        const wrap=document.querySelector('.canvas-wrap'); const w=Math.min(wrap.clientWidth-4,860);
        canvasW=w; canvasH=Math.round(w*0.32); if(canvasH<200) canvasH=200;
        canvas.width=canvasW*2; canvas.height=canvasH*2; canvas.style.width=canvasW+'px'; canvas.style.height=canvasH+'px';

    function resetAll(){

    function updateBeforeDisplay(){
        const p1=mass1*ball1.v, p2=mass2*ball2.v, pt=p1+p2, ek=0.5*mass1*ball1.v**2+0.5*mass2*ball2.v**2;
        elBV1.textContent=ball1.v.toFixed(2)+' m/s'; elBV2.textContent=ball2.v.toFixed(2)+' m/s';
        elBPT.textContent=pt.toFixed(2); elBEk.textContent=ek.toFixed(2)+' J';
    }
    function clearAfter(){ [elAV1,elAV2,elAPT,elAEk,elDV1,elDV2,elDPT,elDEk].forEach(e=>e.textContent='—'); }
    function updateAfter(v1f,v2f){
        const p1f=mass1*v1f, p2f=mass2*v2f, ptf=p1f+p2f, ekf=0.5*mass1*v1f**2+0.5*mass2*v2f**2;
        elAV1.textContent=v1f.toFixed(2)+' m/s'; elAV2.textContent=v2f.toFixed(2)+' m/s';
        elAPT.textContent=ptf.toFixed(2); elAEk.textContent=ekf.toFixed(2)+' J';
        if(beforeData){
            elDV1.textContent=(v1f-beforeData.v1).toFixed(2)+' m/s'; elDV2.textContent=(v2f-beforeData.v2).toFixed(2)+' m/s';
            elDPT.textContent=(ptf-beforeData.pTotal).toFixed(3); elDEk.textContent=(ekf-beforeData.ek).toFixed(2)+' J';
            badge.style.display='inline-block';
            if(Math.abs(ptf-beforeData.pTotal)<0.05){ badge.className='badge badge-ok'; badge.textContent='? 动量守恒'; }
            else { badge.className='badge badge-warn'; badge.textContent='?? 偏差'+Math.abs(ptf-beforeData.pTotal).toFixed(3); }
        }
        if(beforeData) chartData.push({label:'碰撞前',value:beforeData.pTotal});
        chartData.push({label:'碰撞后',value:ptf});
        if(chartData.length>8) chartData.shift();
        drawChart();
    }
    function spawnParticles(px,py){
        for(let i=0;i<20;i++){
            const a=Math.random()*Math.PI*2, s=40+Math.random()*200;
            particles.push({x:px,y:py,vx:Math.cos(a)*s,vy:Math.sin(a)*s-70,life:0.4+Math.random()*0.5,maxLife:0.4+Math.random()*0.5,r:1+Math.random()*3,color:Math.random()<0.5?'#f59e0b':(Math.random()<0.5?'#ef4444':'#3b82f6')});
        }
    }
    function drawScene(){
        ctx.clearRect(0,0,canvasW,canvasH);
        ctx.strokeStyle='rgba(180,190,200,0.2)'; ctx.lineWidth=0.5;
        for(let x=25;x<canvasW;x+=25){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,canvasH);ctx.stroke();}
        const ty=trackY(), lx=toPx(trackLeft), rx=toPx(trackRight);
        ctx.fillStyle='#e8ecf1'; ctx.fillRect(lx-4,ty-6,rx-lx+8,16);
        ctx.strokeStyle='#c0c8d4'; ctx.lineWidth=1.5; ctx.beginPath(); ctx.roundRect(lx-4,ty-6,rx-lx+8,16,6); ctx.stroke();
        function drawCar(px,py,w,h,color,label,v){
            ctx.fillStyle=color; ctx.fillRect(px-w/2,py-h/2,w,h);
            ctx.fillStyle='#333'; ctx.beginPath(); ctx.arc(px-w*0.3,py+h/2,5,0,Math.PI*2); ctx.fill(); ctx.beginPath(); ctx.arc(px+w*0.3,py+h/2,5,0,Math.PI*2); ctx.fill();
            ctx.fillStyle='#fff'; ctx.font='bold 12px Inter'; ctx.textAlign='center'; ctx.fillText(label,px,py-2);
        }
        function drawBullet(px,py,color){ ctx.fillStyle=color; ctx.fillRect(px-6,py-3,12,6); }
        function drawBlock(px,py,w,h,color){ ctx.fillStyle=color; ctx.fillRect(px-w/2,py-h/2,w,h); }
        function drawBall(px,py,r,color,glow,label,v){
            const cr=Math.max(r*SCALE,8);
            ctx.fillStyle=glow+'40'; ctx.beginPath(); ctx.arc(px,py,cr*1.5,0,Math.PI*2); ctx.fill();
            const grad=ctx.createRadialGradient(px-cr*0.3,py-cr*0.35,cr*0.05,px,py,cr);
            grad.addColorStop(0,'#fff'); grad.addColorStop(0.3,color); grad.addColorStop(1,'#1a1a2e');
            ctx.fillStyle=grad; ctx.beginPath(); ctx.arc(px,py,cr,0,Math.PI*2); ctx.fill();
            ctx.fillStyle='#fff'; ctx.font=`bold ${Math.max(cr*0.5,10)}px Inter`; ctx.textAlign='center'; ctx.fillText(label,px,py+3);
        }
        if(scene==='car'){
            const w1=30,h1=20,w2=60,h2=30;
            if(merged && collisionDone){ const mx=ball1.x; drawCar(toPx(mx-mergedOff/2),ty,w1,h1,'#ef4444','轿车',mergedV); drawCar(toPx(mx+mergedOff/2),ty,w2,h2,'#3b82f6','货车',mergedV); }
            else { drawCar(toPx(ball1.x),ty,w1,h1,'#ef4444','轿车',ball1.v); drawCar(toPx(ball2.x),ty,w2,h2,'#3b82f6','货车',ball2.v); }
        } else if(scene==='bullet'){
            if(merged && collisionDone){ drawBlock(toPx(ball1.x),ty,40,30,'#8B5E3C'); drawBullet(toPx(ball1.x),ty,'#333'); }
            else { drawBullet(toPx(ball1.x),ty,'#333'); drawBlock(toPx(ball2.x),ty,40,30,'#8B5E3C'); }
        } else {
            if(merged && collisionDone){ const mx=ball1.x; drawBall(toPx(mx-mergedOff/2),ty,ball1.r,'#ef4444','#fca5a5','1',mergedV); drawBall(toPx(mx+mergedOff/2),ty,ball2.r,'#3b82f6','#93c5fd','2',mergedV); }
            else { drawBall(toPx(ball1.x),ty,ball1.r,'#ef4444','#fca5a5','1',ball1.v); drawBall(toPx(ball2.x),ty,ball2.r,'#3b82f6','#93c5fd','2',ball2.v); }
        }
        for(const p of particles){
            const alpha=Math.max(0,p.life/p.maxLife);
            ctx.fillStyle=`rgba(${p.color==='#ef4444'?'239,68,68':p.color==='#3b82f6'?'59,130,246':'245,158,11'},${alpha})`;
            ctx.beginPath(); ctx.arc(p.x,p.y,p.r*alpha,0,Math.PI*2); ctx.fill();
        }
    }
    function drawChart(){
        const cw=chartCanvas.parentElement.clientWidth-8, ch=180;
        chartCanvas.width=cw*2; chartCanvas.height=ch*2; chartCanvas.style.width=cw+'px'; chartCanvas.style.height=ch+'px';
        const cx=chartCtx; cx.setTransform(2,0,0,2,0,0); cx.clearRect(0,0,cw,ch);
        if(chartData.length===0){ cx.fillStyle='#94a3b8'; cx.font='13px Inter'; cx.textAlign='center'; cx.fillText('运行模拟后显示',cw/2,ch/2); return; }
        const padL=40,padR=20,padT=16,padB=30, pw=cw-padL-padR, ph=ch-padT-padB;
        cx.strokeStyle='#e2e8f0'; cx.beginPath(); cx.moveTo(padL,padT); cx.lineTo(padL,ch-padB); cx.lineTo(cw-padR,ch-padB); cx.stroke();
        const vals=chartData.map(d=>d.value); const minV=Math.min(...vals)-0.5, maxV=Math.max(...vals)+0.5, range=maxV-minV||1;
        if(beforeData){
            const thY=ch-padB-((beforeData.pTotal-minV)/range)*ph;
            cx.strokeStyle='rgba(239,68,68,0.7)'; cx.lineWidth=2; cx.setLineDash([5,3]);
            cx.beginPath(); cx.moveTo(padL,thY); cx.lineTo(cw-padR,thY); cx.stroke(); cx.setLineDash([]);
        }
        cx.strokeStyle='#3b82f6'; cx.lineWidth=2.5; cx.beginPath();
        const pts=chartData.map((d,i)=>({x:padL+(i/(chartData.length-1||1))*pw, y:ch-padB-((d.value-minV)/range)*ph, v:d.value}));
        for(let i=0;i<pts.length;i++){ if(i===0) cx.moveTo(pts[i].x,pts[i].y); else cx.lineTo(pts[i].x,pts[i].y); }
        cx.stroke();
        for(const pt of pts){ cx.fillStyle='#3b82f6'; cx.beginPath(); cx.arc(pt.x,pt.y,4,0,Math.PI*2); cx.fill(); cx.fillStyle='#fff'; cx.beginPath(); cx.arc(pt.x,pt.y,2,0,Math.PI*2); cx.fill(); }
    }
    function animate(ts){
        if(!simRunning && particles.length===0 && !flashText.classList.contains('show')){ drawScene(); animId=null; return; }
        if(lastTime===null) lastTime=ts;
        let dt=(ts-lastTime)/1000*simSpeed; if(dt>0.08) dt=0.08; lastTime=ts;
        if(cooldown>0) cooldown-=dt;
        if(simRunning){
            if(merged){
                const nx=ball1.x+mergedV*dt; const hw=mergedOff/2+Math.max(ball1.r,ball2.r);
                if(nx-hw<trackLeft){ ball1.x=trackLeft+hw; mergedV=Math.abs(mergedV); spawnParticles(toPx(ball1.x),trackY()); }
                else if(nx+hw>trackRight){ ball1.x=trackRight-hw; mergedV=-Math.abs(mergedV); spawnParticles(toPx(ball1.x),trackY()); }
                else ball1.x=nx;
                ball2.x=ball1.x; ball1.v=ball2.v=mergedV;
            } else {
                ball1.x+=ball1.v*dt; ball2.x+=ball2.v*dt;
                [ball1,ball2].forEach(b=>{
                    if(b.x-b.r<trackLeft){ b.x=trackLeft+b.r; b.v=Math.abs(b.v); spawnParticles(toPx(b.x),trackY()); }
                    if(b.x+b.r>trackRight){ b.x=trackRight-b.r; b.v=-Math.abs(b.v); spawnParticles(toPx(b.x),trackY()); }
                });
                const dist=ball2.x-ball1.x, md=ball1.r+ball2.r, relV=ball1.v-ball2.v;
                if(!collisionDone && dist<=md && relV>0.001 && cooldown<=0){
                    collisionDone=true; cooldown=0.5;
                    beforeData={v1:ball1.v,v2:ball2.v,p1:mass1*ball1.v,p2:mass2*ball2.v,pTotal:mass1*ball1.v+mass2*ball2.v,ek:0.5*mass1*ball1.v**2+0.5*mass2*ball2.v**2};
                    const ov=md-dist; if(ov>0){ const tm=mass1+mass2; ball1.x-=ov*(mass2/tm); ball2.x+=ov*(mass1/tm); }
                    let v1f,v2f;
                    if(scene==='balls' && collisionType==='elastic'){
                        v1f=((mass1-mass2)*ball1.v+2*mass2*ball2.v)/(mass1+mass2);
                        v2f=((mass2-mass1)*ball2.v+2*mass1*ball1.v)/(mass1+mass2);
                    } else {
                        v1f=v2f=(mass1*ball1.v+mass2*ball2.v)/(mass1+mass2);
                        merged=true; mergedV=v1f; mergedOff=ball1.r+ball2.r;
                        ball1.x=(ball1.x+ball2.x)/2; ball2.x=ball1.x;
                    }
                    ball1.v=v1f; ball2.v=v2f; updateAfter(v1f,v2f);
                    spawnParticles(toPx((ball1.x+ball2.x)/2),trackY());
                    flashText.classList.remove('show'); void flashText.offsetWidth; flashText.classList.add('show');
                    setTimeout(()=>flashText.classList.remove('show'),700);
                }
            }
            if(!collisionDone) updateBeforeDisplay();
        }
        for(let i=particles.length-1;i>=0;i--){ const p=particles[i]; p.x+=p.vx*dt; p.y+=p.vy*dt; p.vy+=180*dt; p.life-=dt; if(p.life<=0) particles.splice(i,1); }
        drawScene(); animId=requestAnimationFrame(animate);
    }
    function startSim(){
        if(simRunning) return;
        resetAll(); simRunning=true; collisionDone=false; merged=false; mergedV=0; mergedOff=0;
        beforeData=null; particles=[]; cooldown=0; lastTime=null;
        chartData=[{label:'初始',value:mass1*vel1+mass2*vel2}];
        btnStart.textContent='? 运行中...'; btnStart.disabled=true;
        updateBeforeDisplay(); clearAfter(); badge.style.display='none';
        flashText.classList.remove('show'); drawChart();
        if(animId) cancelAnimationFrame(animId); animId=requestAnimationFrame(animate);
    }

    let currentQuiz = null, quizAnswered = false;
    function loadQuiz(quizObj){
        currentQuiz = quizObj || quizBank[Math.floor(Math.random()*quizBank.length)];
        document.getElementById('quizQ').textContent = '? ' + currentQuiz.q;
        const optsDiv = document.getElementById('quizOpts'); optsDiv.innerHTML = '';
        currentQuiz.opts.forEach((opt,i)=>{
            const btn = document.createElement('button'); btn.className='quiz-option';
            btn.textContent = String.fromCharCode(65+i)+'. '+opt;
            btn.addEventListener('click', ()=> handleAnswer(i));
            optsDiv.appendChild(btn);
        });
        document.getElementById('quizFeedback').textContent = '';
        document.getElementById('quizNext').style.display = 'none';
        document.getElementById('quizExtraArea').innerHTML = '';
        quizAnswered = false;
    }
    function handleAnswer(selected){
        if(quizAnswered || !currentQuiz) return;
        quizAnswered = true;
        const opts = document.querySelectorAll('.quiz-option');
        opts.forEach(b=>b.classList.remove('correct','wrong'));
        if(selected === currentQuiz.ans){
            opts[selected].classList.add('correct');
            document.getElementById('quizFeedback').innerHTML = '? 正确！' + currentQuiz.exp;
            document.getElementById('quizNext').style.display = 'inline-block';
        } else {
            opts[selected].classList.add('wrong');
            opts[currentQuiz.ans].classList.add('correct');
            document.getElementById('quizFeedback').innerHTML = '? 错误！' + currentQuiz.exp;
            const sameCategory = quizBank.filter(q=>q.category===currentQuiz.category && q.id!==currentQuiz.id);
            if(sameCategory.length>0){
                const nextQ = sameCategory[Math.floor(Math.random()*sameCategory.length)];
                document.getElementById('quizExtraArea').innerHTML = `<button class="btn btn-preset" id="extraQuizBtn">?? 举一反三：再做一题同类题</button>`;
                document.getElementById('extraQuizBtn').addEventListener('click', ()=> loadQuiz(nextQ));
            }
            document.getElementById('quizNext').style.display = 'inline-block';
        }
    }
    document.getElementById('quizNext').addEventListener('click', ()=> loadQuiz());
