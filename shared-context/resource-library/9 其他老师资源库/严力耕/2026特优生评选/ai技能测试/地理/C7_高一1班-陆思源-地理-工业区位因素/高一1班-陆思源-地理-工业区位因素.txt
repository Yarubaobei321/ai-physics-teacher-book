    backBtn.addEventListener('click', showHome);

    // ========= 区位核心数据 =========
    const industriesData = {
        steel: { name:"钢铁厂", scores:[5,3,2,4,5,2], dominate:"原料·能源·交通", case:"鞍钢依托铁矿，宝钢临海+市场", img:"????" },

        electronics: { name:"电子装配", scores:[1,4,5,4,2,3], dominate:"劳动力+交通", case:"富士康郑州，航空港+百万员工", img:"????" },
        brewery: { name:"啤酒厂", scores:[3,5,2,4,2,3], dominate:"市场", case:"青岛啤酒全国建厂", img:"?????" }
    };
    const factorLabels = ["原料资源","市场距离","劳动力","交通物流","能源动力","政策/土地"];
    let radarChart = null;
    function updateRadar(key) {
        const data = industriesData[key];
        if(!data) return;
        const ctx = document.getElementById('radarChartCanvas').getContext('2d');
        if(radarChart) { radarChart.data.datasets[0].data = data.scores; radarChart.update(); }
        else {
            radarChart = new Chart(ctx, {
                type:'radar', data:{ labels:factorLabels, datasets:[{ label:'重要性 (1-5)', data:data.scores, backgroundColor:'rgba(44,124,182,0.2)', borderColor:'#2c7cb6', borderWidth:2, pointBackgroundColor:'#2c7cb6' }] },
                options:{ responsive:true, maintainAspectRatio:true, scales:{ r:{ beginAtZero:true, max:5, stepSize:1, ticks:{color:'#4f7ea0'}, grid:{color:'#d8e4ef'}, pointLabels:{color:'#1f5a7e'} } } }
            });
        }
        document.getElementById('currentIndustryName').innerHTML = data.name;
        document.getElementById('industryDetailText').innerHTML = `${data.name} ${data.name==='钢铁厂'?'原料+能源导向，现代临海靠市场':'主导区位明显'}`;
        document.getElementById('dominateFactors').innerHTML = data.dominate;
        document.getElementById('realCaseBlock').innerHTML = `<i class="fas fa-globe"></i> <strong>案例：</strong> ${data.case}`;
    }
    const factorKnowledge = {
        raw:{ title:"原料因素", detail:"原料导向型：制糖、钢铁初期；案例：广西糖厂、鞍钢。", img:"?????" },
        market:{ title:"市场因素", detail:"市场导向型：啤酒、家具；青岛啤酒全国设厂。", img:"????" },
        labor:{ title:"劳动力因素", detail:"劳动密集型：服装、电子装配；富士康郑州。", img:"????" },
        transport:{ title:"交通因素", detail:"港口/航空降低运费；上海宝钢、郑州航空港。", img:"????" },
        energy:{ title:"能源动力", detail:"高耗能工业靠近能源地；内蒙古电解铝。", img:"???" },
        policy:{ title:"政策/土地", detail:"特区/新区扶持；深圳特区、特斯拉工厂。", img:"?????" }
    };
    // 因素卡片点击
    document.querySelectorAll('.factor-card').forEach(card => {
        card.addEventListener('click', () => {
            const f = card.dataset.factor;
            if(f && factorKnowledge[f]) showModal(factorKnowledge[f].title, factorKnowledge[f].detail, factorKnowledge[f].img);
            document.querySelectorAll('.factor-card').forEach(c => c.classList.remove('active'));
            card.classList.add('active');
        });
    });
    document.getElementById('industrySelect').addEventListener('change', e => updateRadar(e.target.value));

    // 模态框功能
    const modal = document.getElementById('detailModal');
    const modalTitle = document.getElementById('modalTitle');
    const modalBody = document.getElementById('modalBody');
    const modalImage = document.getElementById('modalImage');
    function showModal(title, detail, imgEmoji = "??") {
        modalTitle.innerText = title;
        modalBody.innerText = detail;
        modalImage.innerHTML = `<div style="font-size:3rem;">${imgEmoji}</div><div style="font-size:0.7rem;">区位示意图</div>`;
        modal.style.display = 'flex';
    }
    document.getElementById('closeModalBtn').addEventListener('click', () => modal.style.display = 'none');
    modal.addEventListener('click', (e) => { if(e.target === modal) modal.style.display = 'none'; });

    // 快问快答
    document.querySelectorAll('.quiz-btn').forEach(btn => {
        btn.addEventListener('click', () => {
            const type = btn.dataset.quiz;
            let ans = "";
            if(type==='sugar') ans = "? 制糖厂首要因素：【原料】! 原料损耗大（8吨甘蔗→1吨糖），原料导向型。";
            else if(type==='brewery') ans = "?? 啤酒厂主导因素：【市场】! 短保质期+高运费，需靠近消费市场。";
            else ans = "?? 电子装配关键：【劳动力+交通】! 富士康郑州依靠航空港+百万劳动力。";
            document.getElementById('quickQuizFeedback').innerHTML = `<i class="fas fa-comment"></i> ${ans}`;
            if(type==='sugar') document.getElementById('industrySelect').value='sugar', updateRadar('sugar');
            if(type==='brewery') document.getElementById('industrySelect').value='brewery', updateRadar('brewery');
            if(type==='electronics') document.getElementById('industrySelect').value='electronics', updateRadar('electronics');
        });
    });

    // 选择题库
    const choiceBank = [
        { text:"下列工业中属于原料导向型的是？", options:["啤酒厂","制糖厂","服装厂","电子装配"], correct:1, explanation:"制糖厂原料损耗大，需接近原料产地。" },
        { text:"德国鲁尔区早期钢铁工业主要依托？", options:["市场","劳动力","煤炭资源","交通"], correct:2, explanation:"鲁尔区丰富的煤炭资源是早期核心优势。" },
        { text:"家具厂趋向大城市周边的主要原因是？", options:["原料","市场运输成本","劳动力","政策"], correct:1, explanation:"家具产品运输成本高，属于市场导向。" }
    ];
    let qIdx = 0;
    function renderChoice() {
        const q = choiceBank[qIdx];
        let html = `<div style="background:#f6fafd; border-radius:24px; padding:16px;"><div style="font-weight:500;">?? ${q.text}</div><div id="choiceOpts" style="margin:10px 0;">`;
        q.options.forEach((opt,i)=> html += `<span class="quiz-option" data-idx="${i}" style="display:inline-block; background:#edf4fa; margin:5px 8px 5px 0; padding:5px 16px; border-radius:40px; cursor:pointer;">${String.fromCharCode(65+i)}. ${opt}</span>`);
        html += `</div><div id="choiceFb" style="margin-top:8px;"></div></div>`;
        document.getElementById('choiceQuestionArea').innerHTML = html;
        document.querySelectorAll('.quiz-option').forEach(el => {
            el.addEventListener('click', () => {
                const selected = parseInt(el.dataset.idx);
                const correct = q.correct;
                const fb = document.getElementById('choiceFb');
                if(selected===correct) fb.innerHTML = `<span style="color:#2c7cb6;">? 正确！ ${q.explanation}</span>`;
                else fb.innerHTML = `<span style="color:#c45a32;">? 错误。正确答案：${String.fromCharCode(65+correct)}. ${q.options[correct]}。${q.explanation}</span>`;
            });
        });
    }
    renderChoice();
    document.getElementById('nextChoiceBtn').addEventListener('click', () => { qIdx = (qIdx+1)%choiceBank.length; renderChoice(); });

    // 易错题集
    const mistakes = [
        { q:"啤酒厂属于原料导向型工业？", correct:"错误。啤酒厂属于市场导向型，产品易变质，运输成本高。" },
        { q:"制糖厂布局主导因素是劳动力？", correct:"错误。主导因素为原料（甘蔗损耗巨大）。" },
        { q:"所有电子工业都是技术导向，无需考虑劳动力？", correct:"错误。电子装配需要大量熟练工，属于劳动力+技术导向。" },
        { q:"钢铁工业永远属于原料导向型？", correct:"现代钢铁工业（如宝钢）依托港口，演变为交通-市场导向。" }
    ];
    document.getElementById('mistakeList').innerHTML = mistakes.map(m => `<div class="question-item"><i class="fas fa-circle-exclamation"></i> ${m.q}<div style="margin-top:6px; color:#2c7cb6;">? 正确答案: ${m.correct}</div></div>`).join('');

    // 高考真题
    const gaokaoData = [{ year:"2022全国乙卷", q:"德国鲁尔区兴起的主要区位因素是？", options:"A.煤炭 B.水运 C.劳动力 D.市场", answer:"A", analysis:"依托丰富的煤炭资源发展钢铁工业。" }];
    document.getElementById('gaokaoContainer').innerHTML = gaokaoData.map(g => `<div class="gaokao-card"><div>?? ${g.year}</div><div>${g.q}</div><div style="font-size:0.8rem;">${g.options}</div><div><span style="background:#eaf3fc; padding:2px 12px; border-radius:30px;">? 答案：${g.answer}</span> ${g.analysis}</div></div>`).join('');

    // 碎片化知识（可点击）
    const knowledgeList = [
        { title:"原料指数", brief:"原料指数 >1 为原料导向型", detail:"原料指数 = 原料重量/产品重量，>1表示原料损耗大，应靠近原料产地。", img:"????" },
        { title:"韦伯工业区位论", brief:"最小运费原理", detail:"韦伯提出工业区位选择考虑运输成本、劳动力成本和集聚效应。", img:"????" },
        { title:"临海型布局", brief:"宝钢、日本太平洋沿岸工业带", detail:"二战后钢铁厂转向沿海港口，利用海运进口原料并接近国际市场。", img:"????" },
        { title:"集聚效应", brief:"共享基础设施、技术外溢", detail:"工业集聚可降低交易成本、促进创新，如硅谷、深圳科技园。", img:"????" }
    ];
    const knowledgeContainer = document.getElementById('knowledgeContainer');
    knowledgeContainer.innerHTML = knowledgeList.map(k => `<div class="knowledge-card" data-knowledge="${k.title}"><i class="fas fa-leaf"></i> <strong>${k.brief}</strong><br><span style="font-size:0.7rem;">点击查看详解 →</span></div>`).join('');
    document.querySelectorAll('.knowledge-card').forEach(card => {
        card.addEventListener('click', () => {
            const title = card.dataset.knowledge;
            const item = knowledgeList.find(k => k.title === title);
            if(item) showModal(item.title, item.detail, item.img);
        });
    });

    // AI命题生成
    function genQuestion(txt) {
        if(!txt.trim()) return { q:"请输入工业案例", a:"例如：制糖厂设在甘蔗产区 → 原料导向型", img:"?" };
        const low = txt.toLowerCase();
        if(low.includes("制糖")) return { q:`根据“${txt}”，主导区位是？A.原料 B.市场 C.劳动力`, a:"正确答案：A (原料导向型)", img:"????" };
        if(low.includes("啤酒")||low.includes("家具")) return { q:`“${txt}”最可能的主导区位？A.原料 B.市场 C.劳动力`, a:"正确答案：B (市场导向型)", img:"????" };
        if(low.includes("电解铝")||low.includes("高耗能")) return { q:`“${txt}”主导因素？A.原料 B.能源 C.市场`, a:"正确答案：B (能源导向型)", img:"???" };
        if(low.includes("电子")||low.includes("富士康")) return { q:`“${txt}”工业区位主要考虑？A.劳动力+交通 B.原料 C.市场`, a:"正确答案：A (劳动力与交通)", img:"????" };
        return { q:`区位分析：“${txt}”优势因素？A.原料 B.市场 C.劳动力 D.交通`, a:"参考答案：结合具体条件，通常考虑运输成本或资源禀赋。", img:"????" };
    }
    document.getElementById('generateQuestionBtn').addEventListener('click', () => {
        const input = document.getElementById('userIndustryInput').value;
        const {q,a,img} = genQuestion(input);
        const area = document.getElementById('generatedQuestionArea');
        area.style.display = 'block';
        area.innerHTML = `<div><i class="fas fa-question-circle"></i> <strong>题目：</strong><br>${q}</div><div style="margin-top:12px; background:#e6f0f9; border-radius:18px; padding:8px;"><i class="fas fa-check-circle"></i> <strong>正确答案：</strong><br>${a}</div><div style="margin-top:8px; font-size:2rem;">${img}</div>`;
    });

    // AI悬浮聊天（陆固安）
    const aiFab = document.getElementById('aiFab');
    const aiPanel = document.getElementById('aiPanel');
    const closeAi = document.getElementById('closeAiPanel');
    aiFab.addEventListener('click', () => aiPanel.classList.toggle('show'));
    closeAi.addEventListener('click', () => aiPanel.classList.remove('show'));
    const aiChat = document.getElementById('aiChatMessages');
    const aiInput = document.getElementById('aiChatInput');
    const aiSend = document.getElementById('aiSendBtn');
    function addAiMsg(text, isUser) {
        const div = document.createElement('div');
        div.className = `ai-message ${isUser ? 'ai-user' : 'ai-bot'}`;
        div.innerHTML = isUser ? `<i class="fas fa-user"></i> ${text}` : `<i class="fas fa-robot"></i> ${text}`;
        aiChat.appendChild(div);
        aiChat.scrollTop = aiChat.scrollHeight;
    }
    function aiReply(question) {
        const q = question.toLowerCase();
        if(q.includes("原料导向")) return "原料导向型工业：原料损耗大或运输成本高，应接近原料产地。如制糖厂。";
        if(q.includes("市场导向")) return "市场导向型工业：产品不便储存或运输成本高，需靠近消费市场。如啤酒厂。";
        if(q.includes("劳动力导向")) return "劳动力导向型工业需要大量廉价或熟练劳动力，如服装厂、电子装配厂。";
        if(q.includes("钢铁区位")||q.includes("钢铁演变")) return "钢铁工业区位演变：早期靠近煤铁资源（原料导向），现代转向沿海港口（交通/市场导向）。";
        if(q.includes("富士康")) return "富士康郑州选址优势：①河南劳动力丰富，②郑州航空港物流，③政策支持。";
        if(q.includes("啤酒厂")) return "啤酒厂属于市场导向型，因保质期短、运输成本高，在大城市周边建厂。";
        if(q.includes("生成题目:")) {
            const material = question.replace("生成题目:", "").trim();
            const {q:genQ, a:genA} = genQuestion(material);
            return `?? 生成题目：${genQ}\n? 答案：${genA}`;
        }
        return "我是陆固安，你可以问：“什么是原料导向型？”“啤酒厂为什么靠近市场？”或输入“生成题目: 电解铝厂”来命题。";
    }
    function handleAiChat() {
        const text = aiInput.value.trim();
        if(!text) return;
        addAiMsg(text, true);
        aiInput.value = '';
        setTimeout(() => addAiMsg(aiReply(text), false), 200);
    }
    aiSend.addEventListener('click', handleAiChat);
    aiInput.addEventListener('keypress', e => { if(e.key === 'Enter') handleAiChat(); });
    document.querySelectorAll('.ai-prompt-btn').forEach(btn => {
        btn.addEventListener('click', () => {
            const prompt = btn.dataset.prompt;
            aiInput.value = prompt;
            handleAiChat();
        });
    });

    updateRadar('steel');
</script>
